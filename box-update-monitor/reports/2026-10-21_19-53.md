```diff
+}
+
+old_pid="$(tr -dc '0-9' 2>/dev/null <"${PID_FILE}" || true)"
+if [ -n "${old_pid}" ] && [ "${old_pid}" != "$$" ] &&
+	predecessor_is_keyring "${old_pid}"; then
+	kill "${old_pid}" 2>/dev/null || true
+	for _ in $(seq 1 50); do
+		kill -0 "${old_pid}" 2>/dev/null || break
+		sleep 0.1
+	done
+	if kill -0 "${old_pid}" 2>/dev/null; then
+		kill -9 "${old_pid}" 2>/dev/null || true
+	fi
+fi
+
+if [ ! -s "${SECRET_FILE}" ]; then
+	# A login keyring restored without its unlock secret (a partial sync can
+	# save the manifest after the secret upload failed) can never be unlocked
+	# again: a freshly minted secret would leave the daemon prompting on every
+	# access and the collection permanently locked. Move the orphan aside so
+	# the daemon creates a fresh, unlockable keyring; the old bytes stay on
+	# disk for manual recovery. Only login.keyring — it is the one keyring
+	# this script's secret unlocks; other keyrings have their own passwords.
+	if [ -e "${KEYRINGS_DIR}/login.keyring" ]; then
+		mv -f "${KEYRINGS_DIR}/login.keyring" \
+			"${KEYRINGS_DIR}/login.keyring.orphaned.$(date +%s)"
+	fi
+	head -c 32 /dev/urandom | base64 >"${SECRET_FILE}.tmp"
+	chmod 600 "${SECRET_FILE}.tmp"
+	mv -f "${SECRET_FILE}.tmp" "${SECRET_FILE}"
+fi
+
+# Point the `default` collection alias at `login` unless it already names a
+# keyring that exists on disk. A dangling alias (the image ships none, and a
+# session-keyring-only box leaves a `default` file naming nothing durable)
+# would keep clients on the in-memory session collection.
+DEFAULT_FILE="${KEYRINGS_DIR}/default"
+current_default=""
+if [ -f "${DEFAULT_FILE}" ]; then
+	current_default="$(tr -d '[:space:]' <"${DEFAULT_FILE}")"
+fi
+if [ "${current_default}" != "login" ] &&
+	{ [ -z "${current_default}" ] || [ ! -f "${KEYRINGS_DIR}/${current_default}.keyring" ]; }; then
+	printf 'login' >"${DEFAULT_FILE}"
+fi
+
+# --unlock reads the password from stdin and unlocks the `login` keyring,
+# creating it when it does not exist yet (gnome-keyring-daemon(1)). secrets is
+# the only component the box needs; ssh/pkcs11 stay off. exec preserves this
+# pid into the daemon, so the pid recorded here is the one a successor
+# replaces and the one the desktop supervisor watches.
+printf '%s' "$$" >"${PID_FILE}"
+exec 9>&-
+exec gnome-keyring-daemon --foreground --components=secrets --unlock <"${SECRET_FILE}"
```
